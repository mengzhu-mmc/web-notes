# ahooks 源码级学习指南 (中)：生命周期与状态增强

## 三、生命周期与 Effect 管理

### 6. `useMount` & `useUnmount`

- **作用：** 组件挂载/卸载时执行。
- **核心源码实现：**

```tsx
function useMount(fn: () => void) {
  // 注意开发环境 StrictMode 下会执行两次
  useEffect(() => {
    fn?.();
  }, []);
}

function useUnmount(fn: () => void) {
  // 核心：使用 useLatest 保证在卸载时，内部一定能拿到最新的 state
  const fnRef = useLatest(fn);

  useEffect(
    () => () => {
      fnRef.current();
    },
    [],
  );
}
```

### 7. `useUpdateEffect`

- **作用：** 忽略首次渲染，只在依赖更新时执行（类似 Vue watch）。
- **核心源码实现：**

```tsx
function useUpdateEffect(effect: EffectCallback, deps?: DependencyList) {
  const isMounted = useRef(false);

  useEffect(() => {
    return () => {
      isMounted.current = false;
    };
  }, []);

  useEffect(() => {
    // 首次渲染直接跳过
    if (!isMounted.current) {
      isMounted.current = true;
    } else {
      return effect();
    }
  }, deps);
}
```

### 8. `useDeepCompareEffect`

- **作用：** 用法与 `useEffect` 一致，但依赖数组采用深度比较。
- **核心源码实现：**

```tsx
import isEqual from "lodash/isEqual";

function useDeepCompareEffect(effect: EffectCallback, deps: DependencyList) {
  const ref = useRef<DependencyList>();
  const signalRef = useRef<number>(0); // 内部信号量

  // 1. 如果依赖没变（深度比较），信号量不变
  // 2. 如果变了，让信号量 +1
  if (!isEqual(deps, ref.current)) {
    ref.current = deps;
    signalRef.current += 1;
  }

  // 把内部递增的信号量作为真实 useEffect 的依赖
  useEffect(effect, [signalRef.current]);
}
```

### 9. `useUpdate`

- **作用：** 返回一个函数，调用强制组件重新渲染。
- **核心源码实现：**

```tsx
function useUpdate() {
  const [, setState] = useState({});
  // 利用空对象 {} 每次内存地址不同的特性触发 render
  return useCallback(() => setState({}), []);
}
```

## 四、状态管理增强

### 10. `useSetState`

- **作用：** 管理对象状态，支持类似 Class 组件的自动合并。
- **核心源码实现：**

```tsx
function useSetState<S extends Record<string, any>>(
  initialState: S | (() => S),
) {
  const [state, setState] = useState<S>(initialState);

  const setMergeState = useCallback((patch) => {
    setState((prevState) => {
      const newState = typeof patch === "function" ? patch(prevState) : patch;
      // 核心：使用 Object.assign 合并对象
      return newState ? Object.assign({}, prevState, newState) : prevState;
    });
  }, []);

  return [state, setMergeState];
}
```

### 11. `useBoolean` / `useToggle`

- **作用：** 优雅地管理布尔值或两个状态切换。
- **核心源码实现 (useToggle)：**

```tsx
function useToggle<T = boolean, U = boolean>(
  defaultValue: T = false as unknown as T,
  reverseValue?: U,
) {
  const [state, setState] = useState<T | U>(defaultValue);

  const actions = useMemo(() => {
    const reverse = (
      reverseValue === undefined ? !defaultValue : reverseValue
    ) as T | U;

    return {
      toggle: () =>
        setState((s) => (s === defaultValue ? reverse : defaultValue)),
      set: (value: T | U) => setState(value),
      setLeft: () => setState(defaultValue),
      setRight: () => setState(reverse),
    };
  }, [defaultValue, reverseValue]);

  return [state, actions];
}
```

### 12. `usePrevious`

- **作用：** 保存状态的上一次的值。
- **核心源码实现：**

```tsx
function usePrevious<T>(state: T): T | undefined {
  const prevRef = useRef<T>();
  const curRef = useRef<T>();

  // 1. 如果新传入的值和 curRef 不相等
  if (!Object.is(curRef.current, state)) {
    // 2. 把当前的 curRef 降级为 prevRef
    prevRef.current = curRef.current;
    // 3. 把新值赋给 curRef
    curRef.current = state;
  }

  // 永远返回旧值的盒子
  return prevRef.current;
}
```

### 13. `useGetState`

- **作用：** 给 `useState` 增加了一个 `getter` 方法，避开闭包陷阱读取最新状态。
- **核心源码实现：**

```tsx
function useGetState<S>(initialState: S) {
  const [state, setState] = useState<S>(initialState);
  const stateRef = useRef<S>(state);

  // 关键：每次更新时，不仅 setState，还要同步到 ref
  stateRef.current = state;

  const getState = useCallback(() => stateRef.current, []);

  return [state, setState, getState];
}
```
