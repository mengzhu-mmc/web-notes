### 1. SharedWorker - Baseline 2026 新可用
> 允许从多个浏览上下文（多个窗口或 iframe）访问同一个 worker，实现跨窗口/标签页的通信和资源共享

**浏览器支持**：Chrome 58+ / Edge 79+ / Firefox 105+ / Safari 15.4+ (2026 年 5 月起成为 Baseline 新可用特性)

**代码示例**
`javascript
// 主线程 - 页面1
const myWorker = new SharedWorker("worker.js");
myWorker.port.start();

myWorker.port.postMessage([10, 20]);

myWorker.port.onmessage = (e) => {
  console.log("Result from worker:", e.data);
};

// 主线程 - 页面2（同一 worker）
const myWorker2 = new SharedWorker("worker.js");
myWorker2.port.start();

myWorker2.port.postMessage([30, 40]);

// worker.js - 共享的 worker 线程
onconnect = (e) => {
  const port = e.ports[0];
  
  port.onmessage = (e) => {
    const result = e.data[0] * e.data[1];
    port.postMessage(result);
  };
  
  port.start();
};
`

### 2. ECMAScript 2026 - using 关键字（显式资源管理）

**代码示例**
`javascript
class DatabaseConnection {
  constructor(name) {
    this.name = name;
    console.log(Database  connected.);
  }
  
  query(sql) {
    return Executing '' on ;
  }
  
  [Symbol.dispose]() {
    console.log(Database  closed.);
  }
}

function manageResource() {
  using db = new DatabaseConnection("MySQL");
  
  // 使用资源
  const result = db.query("SELECT * FROM users");
  console.log(result);
  
} // 这里自动调用 Symbol.dispose()，关闭数据库连接

manageResource();
// 输出：
// Database MySQL connected.
// Executing 'SELECT * FROM users' on MySQL
// Database MySQL closed.

### 3. ECMAScript 2026 - Array.fromAsync()
> 从异步可迭代对象、类数组对象或 Promise 创建新数组，简化异步数据处理

**浏览器支持**：Chrome 121+ / Edge 121+ / Firefox 115+ / Safari 16.4+ (Opera 不支持)

**代码示例**
`javascript
// 从异步生成器创建数组
async function* asyncGenerator() {
  yield Promise.resolve(1);
  yield Promise.resolve(2);
  yield Promise.resolve(3);
}

async function processAsyncData() {
  const arr = await Array.fromAsync(asyncGenerator());
  console.log(arr); // [1, 2, 3]
}

processAsyncData();

// 从异步可迭代对象创建数组
const asyncIterable = {
  async *[Symbol.asyncIterator]() {
    for (let i = 0; i < 5; i++) {
      yield await Promise.resolve(i);
    }
  }
};

async function main() {
  const result = await Array.fromAsync(asyncIterable);
  console.log(result); // [0, 1, 2, 3, 4]
}

main();

### 4. ECMAScript 2026 - Uint8Array Base64 和 Hex 编码
> 直接在 Uint8Array 和 Base64/Hex 字符串之间转换，无需手动编码/解码

**浏览器支持**：Chrome 140+ / Edge 140+ / Firefox 133+ / Safari 18.2+ (Opera 不支持)

**代码示例**

console.log(arrFromBase64); // Uint8Array [72, 101, 108, 108, 111, 32, 87, 111, 114, 108, 100]

### 5. ECMAScript 2026 - Error.isError()
> 可靠地检查一个值是否为 Error 对象，解决跨 Realm 环境下 instanceof Error 失效的问题

**浏览器支持**：Chrome 134+ / Edge 134+ / Firefox 138+ / Opera 119+ (Safari 不支持)

**代码示例**
`javascript
// 传统方法的问题：跨 Realm 失效
const iframe = document.createElement('iframe');
document.body.appendChild(iframe);
const iframeError = new iframe.contentWindow.Error('Error from iframe');

console.log(iframeError instanceof Error); // false（跨 Realm 失效）
console.log(Error.isError(iframeError)); // true（可靠）

// 正常使用
console.log(Error.isError(new TypeError())); // true
console.log(Error.isError(new RangeError())); // true
console.log(Error.isError(new Error())); // true

// 非 Error 对象
console.log(Error.isError({ name: 'Error' })); // false
console.log(Error.isError('Error')); // false
console.log(Error.isError(null)); // false
`
